"""Normal owner step-up revoke of the single dedicated T0224 test rule."""

import json, ssl
from pathlib import Path
from datetime import datetime, timezone
import httpx

s = json.loads(Path("/state/fixtures.json").read_text())
password = Path("/admin-password").read_text().strip()
url = "https://100.72.60.100:8443"
with httpx.Client(
    base_url=url,
    verify=ssl.create_default_context(cafile="/ca.crt"),
    timeout=20,
    headers={"Origin": url},
) as c:
    r = c.post(
        "/api/v1/auth/login",
        json={"email": "network-acceptance@example.com", "password": password},
    )
    r.raise_for_status()
    path = (
        "/api/v1/runner/environments/"
        + s["environments"]["target"]["id"]
        + "/ingress/"
        + s["ingress"]["id"]
        + "/revoke"
    )
    r = c.post(
        path,
        headers={
            "X-Workspace-Id": s["workspace"],
            "X-CSRF-Token": c.cookies.get("ws_csrf") or "",
        },
        json={"password": password},
    )
    r.raise_for_status()
    value = r.json()
    assert value["state"] == "revoked"
    Path("/state/revoke-result.json").write_text(
        json.dumps(
            {
                "utc": datetime.now(timezone.utc).isoformat(),
                "record_id": value["id"],
                "revision": value["desired_revision"],
                "state": value["state"],
            }
        )
    )
    print("dedicated test rule revoked through normal owner step-up API")
